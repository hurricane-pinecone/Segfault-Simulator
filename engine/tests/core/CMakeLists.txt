# engine-core tests: each is a standalone executable linking engine-core (no
# third-party deps) and registered with CTest. add_core_test(name source)
# declares one in a line.
function(add_core_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE engine-core)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS core)
endfunction()

add_core_test(luaTests scripting/luaTests.cpp)
add_core_test(ecsTests ecs/ecsTests.cpp)
add_core_test(ecsBenchTests ecs/ecsBenchTests.cpp)
# Tests that declare event types get their compile-time traits, as a game
# module does (sfs_event_traits). Without a codegen to write them (a core-only
# build given no SFS_CODEGEN_EXECUTABLE) they're left out.
function(add_core_event_test name source)
    if(NOT SFS_CODEGEN_COMMAND)
        return()
    endif()
    add_core_test(${name} ${source})
    sfs_event_traits(${name} FILES "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endfunction()
if(NOT SFS_CODEGEN_COMMAND)
    message(STATUS "No sfs-codegen: the event tests are left out (pass "
                   "SFS_CODEGEN_EXECUTABLE=<native-build>/bin/sfs-codegen)")
endif()

# Compile-fail tests: each case of eventCompileFail.cpp is its own target,
# left out of the build, which the test builds and expects to fail with its
# static_assert's message (or the failing template's name, which a message
# can't hold). Case 0 holds the allowed uses and must build.
function(add_event_compile_test number expected)
    if(NOT SFS_CODEGEN_COMMAND)
        return()
    endif()
    set(target eventCompileFail${number})
    add_library(${target} OBJECT EXCLUDE_FROM_ALL events/eventCompileFail.cpp)
    target_link_libraries(${target} PRIVATE engine-core)
    target_compile_definitions(${target} PRIVATE SFS_COMPILE_FAIL_CASE=${number})
    sfs_event_traits(${target} FILES
        "${CMAKE_CURRENT_SOURCE_DIR}/events/eventCompileFail.cpp")
    add_test(NAME ${target}
             COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}
                     --target ${target} --config $<CONFIG>)
    set_tests_properties(${target} PROPERTIES LABELS core
                         RESOURCE_LOCK eventCompileFail)
    if(number EQUAL 0)
        return()
    endif()
    set_tests_properties(${target} PROPERTIES
                         PASS_REGULAR_EXPRESSION "${expected}")
endfunction()
