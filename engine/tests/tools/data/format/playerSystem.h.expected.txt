#pragma once

#include <sfs/sfs.h>

#include <vector>

// Moves the players.
class PlayerSystem : public sfs::System
{
public:
  /** A player's state. */
  struct State
  {
    void reset();

    int health = 0;
    float speed = 0.0f;
  };

  /** Makes one. */
  explicit PlayerSystem(float speed);
  ~PlayerSystem() override;

  /** Adds a player. */
  void add(int id);
  /** Adds a player at a place. */
  void add(int id, float x);
  [[nodiscard]] bool empty() const { return m_count == 0; }
  /** The players' top speed. */
  float topSpeed() const;

protected:
  void create() override;
  void update(double deltaTime) override;

private:
  template <typename T>
  T* find(int id);
  void step(State& state);

  // Scratch, reused each frame.
  std::vector<int> m_scratch;
  int m_count = 0; // how many there are
  [[sfs::hide]]
  float m_speed = 1.0f;
};
